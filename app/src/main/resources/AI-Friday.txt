# 🔎 vector_db.py

```python
from qdrant_client import QdrantClient
from qdrant_client.models import VectorParams, Distance
from config import QDRANT_HOST

client = QdrantClient(host=QDRANT_HOST, port=6333)

COLLECTION = "code_vectors"

client.recreate_collection(
    collection_name=COLLECTION,
    vectors_config=VectorParams(size=384, distance=Distance.COSINE)
)

def add_vector(id, vector, payload):
    client.upsert(
        collection_name=COLLECTION,
        points=[{"id": id, "vector": vector, "payload": payload}]
    )

def search_vector(vector):
    return client.search(
        collection_name=COLLECTION,
        query_vector=vector,
        limit=5
    )
```

# 🤖 agents/ingestion_agent.py

```python
import os, git
from services.chunker import chunk
from services.embedding_service import embed
from vector_db import add_vector

def ingest_repo(repo_url):
    if os.path.exists("repo"):
        import shutil
        shutil.rmtree("repo")

    git.Repo.clone_from(repo_url, "repo")

    idx = 0
    for root, _, files in os.walk("repo"):
        for f in files:
            if f.endswith((".py", ".java", ".scala")):
                path = os.path.join(root, f)
                with open(path, "r", errors="ignore") as file:
                    content = file.read()
                    chunks = chunk(content)

                    for c in chunks:
                        vector = embed(c)
                        add_vector(idx, vector, {"file": path, "code": c})
                        idx += 1
```

# 🧠 services/vector_store.py

```python
import faiss
import numpy as np

dimension = 384
index = faiss.IndexFlatL2(dimension)
metadata = []

def add_vector(vector, payload):
    vec = np.array([vector]).astype("float32")
    index.add(vec)
    metadata.append(payload)

def search_vector(vector, k=5):
    vec = np.array([vector]).astype("float32")
    D, I = index.search(vec, k)

    results = []
    for idx in I[0]:
        if idx < len(metadata):
            results.append(metadata[idx])
    return results
```

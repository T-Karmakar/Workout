# 🔎 vector_db.py

```python
from qdrant_client import QdrantClient
from qdrant_client.models import VectorParams, Distance
from config import QDRANT_HOST

client = QdrantClient(host=QDRANT_HOST, port=6333)

COLLECTION = "code_vectors"

client.recreate_collection(
    collection_name=COLLECTION,
    vectors_config=VectorParams(size=384, distance=Distance.COSINE)
)

def add_vector(id, vector, payload):
    client.upsert(
        collection_name=COLLECTION,
        points=[{"id": id, "vector": vector, "payload": payload}]
    )

def search_vector(vector):
    return client.search(
        collection_name=COLLECTION,
        query_vector=vector,
        limit=5
    )
```

```python
import os, git
from services.chunker import chunk
from services.embedding_service import embed
from vector_db import add_vector

def ingest_repo(repo_url):
    if os.path.exists("repo"):
        import shutil
        shutil.rmtree("repo")

    git.Repo.clone_from(repo_url, "repo")

    idx = 0
    for root, _, files in os.walk("repo"):
        for f in files:
            if f.endswith((".py", ".java", ".scala")):
                path = os.path.join(root, f)
                with open(path, "r", errors="ignore") as file:
                    content = file.read()
                    chunks = chunk(content)

                    for c in chunks:
                        vector = embed(c)
                        add_vector(idx, vector, {"file": path, "code": c})
                        idx += 1
```

# 🧠 services/vector_store.py

```python
import faiss
import numpy as np

dimension = 384
index = faiss.IndexFlatL2(dimension)
metadata = []

def add_vector(vector, payload):
    vec = np.array([vector]).astype("float32")
    index.add(vec)
    metadata.append(payload)

def search_vector(vector, k=5):
    vec = np.array([vector]).astype("float32")
    D, I = index.search(vec, k)

    results = []
    for idx in I[0]:
        if idx < len(metadata):
            results.append(metadata[idx])
    return results
```

```python
client = OpenAI(api_key=OPENAI_API_KEY)

def ask_llm(prompt, temperature=0.1):
    response = client.chat.completions.create(
        model=LLM_MODEL,
        messages=[{"role": "user", "content": prompt}],
        temperature=temperature
    )
    return response.choices[0].message.content
```

# 🧪 1️⃣ Synthetic Data Generator

## backend/synthetic_data.py

```python
import random
from faker import Faker

fake = Faker()

DIAGNOSES = [
    "Type 2 Diabetes",
    "Breast Cancer Stage II",
    "Hypertension",
    "Rheumatoid Arthritis",
    "Chronic Kidney Disease",
    "Asthma"
]

MEDICATIONS = [
    "Metformin",
    "Insulin",
    "Lisinopril",
    "Methotrexate",
    "Atorvastatin"
]

def generate_patient():
    age = random.randint(18, 85)
    diagnosis = random.choice(DIAGNOSES)

    labs = {
        "HbA1c": round(random.uniform(5.5, 10.0), 1),
        "Creatinine": round(random.uniform(0.6, 2.5), 2),
        "Blood Pressure": f"{random.randint(100,160)}/{random.randint(60,100)}"
    }

    patient = f"""
Patient Name: {fake.name()}
Age: {age}
Gender: {random.choice(['Male','Female'])}
Diagnosis: {diagnosis}
Medications: {random.choice(MEDICATIONS)}
Lab Results:
- HbA1c: {labs['HbA1c']}
- Creatinine: {labs['Creatinine']}
- Blood Pressure: {labs['Blood Pressure']}
"""
    return patient


def generate_trial():
    min_age = random.randint(18, 50)
    max_age = random.randint(60, 85)
    required_diagnosis = random.choice(DIAGNOSES)

    trial = f"""
Inclusion Criteria:
- Age between {min_age} and {max_age}
- Diagnosis of {required_diagnosis}

Exclusion Criteria:
- Creatinine > 2.0
- Uncontrolled hypertension
"""
    return trial
```

```python
from fastapi import FastAPI
from synthetic_data import generate_patient, generate_trial
from agents.patient_agent import extract_patient
from agents.trial_agent import parse_trial
from agents.matching_agent import match
from agents.explanation_agent import explain

app = FastAPI(title="AI Clinical Trial Eligibility Screener")

@app.get("/generate")
def generate():
    patient = generate_patient()
    trial = generate_trial()
    return {"patient": patient, "trial": trial}

@app.post("/screen")
def screen(patient_text: str, trial_text: str):
    patient_structured = extract_patient(patient_text)
    trial_structured = parse_trial(trial_text)
    match_result = match(patient_structured, trial_structured)
    explanation = explain(match_result)

    return {
        "match": match_result,
        "explanation": explanation
    }
```

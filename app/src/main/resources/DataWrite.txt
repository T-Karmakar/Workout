import org.apache.spark.sql.Dataset
import org.apache.spark.sql.SparkSession

object WriteLargeStringArray {
  def main(args: Array[String]): Unit = {
    val spark = SparkSession.builder()
      .appName("WriteLargeStringArray")
      // Add any other Spark configurations as needed
      .master("local[*]") // Use a proper master in a cluster environment
      .getOrCreate()

    import spark.implicits._

    // Assuming your Dataset[Row] is named 'inputDataset' and the column
    // containing the large string is named 'stringValue'

    // Sample data (replace with your actual Dataset)
    val data = Seq(
      Row("[value1,value2,value3]"),
      Row("[another,set,of,values]")
    )
    val inputDataset: Dataset[Row] = spark.createDataFrame(data, new org.apache.spark.sql.types.StructType().add("stringValue", org.apache.spark.sql.types.StringType))

    val outputFilePath = "output.dat"

    // Efficiently transform the data and write to file
    inputDataset.rdd.map { row =>
      val stringValue = row.getString(0)
      stringValue.stripPrefix("[").stripSuffix("]") + "\n"
    }.saveAsTextFile(outputFilePath)

    spark.stop()
  }
}

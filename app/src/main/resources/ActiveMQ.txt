```scala
import org.apache.spark.sql.{SparkSession, DataFrame}
import java.util.Properties

object MultiDatabaseSparkApp {
  def main(args: Array[String]): Unit = {
    val spark = SparkSession.builder()
      .appName("Multi-Database JDBC Example")
      .master("local[*]")  // Use appropriate cluster config in production
      .getOrCreate()

    // --- Database 1: MySQL ---
    val db1Url = "jdbc:mysql://host1:3306/database1"
    val db1Props = new Properties()
    db1Props.setProperty("user", "db1_user")
    db1Props.setProperty("password", "db1_password")
    db1Props.setProperty("driver", "com.mysql.cj.jdbc.Driver")

    val df1: DataFrame = spark.read
      .option("partitionColumn", "id")
      .option("lowerBound", "1")
      .option("upperBound", "100000")
      .option("numPartitions", "4")
      .jdbc(db1Url, "table1", db1Props)

    // --- Database 2: PostgreSQL ---
    val db2Url = "jdbc:postgresql://host2:5432/database2"
    val db2Props = new Properties()
    db2Props.setProperty("user", "db2_user")
    db2Props.setProperty("password", "db2_password")
    db2Props.setProperty("driver", "org.postgresql.Driver")

    val df2: DataFrame = spark.read
      .option("partitionColumn", "ref_id")
      .option("lowerBound", "1")
      .option("upperBound", "100000")
      .option("numPartitions", "4")
      .jdbc(db2Url, "table2", db2Props)

    // --- Join the DataFrames ---
    val joinedDf = df1.join(df2, df1("id") === df2("ref_id"))
    joinedDf.show(20)
```

## ✅ Final Reusable Function with Logging & Error Handling

```scala
import org.apache.activemq.ActiveMQConnectionFactory
import org.apache.activemq.jms.pool.PooledConnectionFactory
import javax.jms._
import com.fasterxml.jackson.databind.ObjectMapper
import org.apache.spark.rdd.RDD
import org.apache.spark.{SparkConf, SparkContext}
import org.slf4j.LoggerFactory

// Message case class
case class MyMessage(id: Int, content: String)

object ActiveMQBatchSender {

  private val logger = LoggerFactory.getLogger("ActiveMQBatchSender")
  private val objectMapper = new ObjectMapper()

  def sendToActiveMQ(
      rdd: RDD[MyMessage],
      queueName: String,
      brokerUrl: String,
      batchSize: Int = 1000,
      maxConnections: Int = 10
  ): Unit = {

    val connectionFactory = new ActiveMQConnectionFactory(brokerUrl)
    val pooledFactory = new PooledConnectionFactory()
    pooledFactory.setConnectionFactory(connectionFactory)
    pooledFactory.setMaxConnections(maxConnections)

    val sc = rdd.context
    val broadcastFactory = sc.broadcast(pooledFactory)

    rdd.foreachPartition { partition =>
      val partitionId = TaskContext.getPartitionId()
      var sentCount = 0
      var batchCount = 0

      try {
        val connection = broadcastFactory.value.createConnection()
        connection.start()

        val session = connection.createSession(false, Session.AUTO_ACKNOWLEDGE)
        val destination = session.createQueue(queueName)
        val producer = session.createProducer(destination)
        producer.setDeliveryMode(DeliveryMode.NON_PERSISTENT)

        partition.grouped(batchSize).foreach { batch =>
          try {
            batch.foreach { msg =>
              val json = objectMapper.writeValueAsString(msg)
              val message = session.createTextMessage(json)
              producer.send(message)
              sentCount += 1
            }
            batchCount += 1
          } catch {
            case e: Exception =>
              logger.error(s"[Partition $partitionId] Error sending batch: ${e.getMessage}", e)
          }
        }

        logger.info(s"[Partition $partitionId] Sent $sentCount messages in $batchCount batches.")

        producer.close()
        session.close()
        connection.close()
      } catch {
        case e: Exception =>
          logger.error(s"[Partition $partitionId] Error setting up JMS connection: ${e.getMessage}", e)
      }
    }
  }
}
```

import org.apache.spark.SparkConf
import scala.concurrent.{ExecutionContext, Future}
import java.util.concurrent.Executors
import scala.util.{Success, Failure}
import scala.collection.mutable.ListBuffer

object FastAsyncActiveMQProducer {
  def main(args: Array[String]): Unit = {
    val conf = new SparkConf().setAppName("FastAsyncActiveMQProducer").setMaster("local[*]") // Adjust
    val sc = new SparkContext(conf)

    val activeMQBrokerURL = "tcp://localhost:61616" // Replace
    val queueName = "fastAsyncQueue"
    val numberOfMessages = 50000
    val numPartitions = sc.defaultParallelism * 4 // Increase parallelism
    val batchSize = 500 // Tune this based on message size and broker capacity
    val threadsPerTask = 10 // Tune this for async sends within each task

    val messagesRDD = sc.parallelize(1 to numberOfMessages, numPartitions)

    messagesRDD.foreachPartitionAsync { partition =>
      val connectionFactory = new ActiveMQConnectionFactory(activeMQBrokerURL)
      var connection: javax.jms.Connection = null
      var session: Session = null
      var producer: MessageProducer = null
      val executorService = Executors.newFixedThreadPool(threadsPerTask)
      implicit val ec = ExecutionContext.fromExecutor(executorService)

      try {
        connection = connectionFactory.createConnection()
        connection.start()
        session = connection.createSession(false, Session.AUTO_ACKNOWLEDGE)
        val destination = session.createQueue(queueName)
        producer = session.createProducer(destination)

        val messageBuffer = new ListBuffer[TextMessage]()

        partition.foreach { i =>
          val messageText = s"Fast Async Message $i" // Keep message size small for the target
          val message = session.createTextMessage(messageText)
          messageBuffer += message

          if (messageBuffer.size >= batchSize) {
            val batchToSend = messageBuffer.toList
            messageBuffer.clear()
            Future {
              batchToSend.foreach(producer.send)
            }
          }
        }

        // Send any remaining messages in the buffer
        if (messageBuffer.nonEmpty) {
          val remainingBatch = messageBuffer.toList
          Future {
            remainingBatch.foreach(producer.send)
          }
        }

        executorService.shutdown()
      } catch {
        case e: Exception => println(s"Error in partition: ${e.getMessage}")
        if (executorService != null) executorService.shutdownNow()
      } finally {
        if (producer != null) producer.close()
        if (session != null) session.close()
        if (connection != null) connection.close()
      }
    }

    // Keep main thread alive
    Thread.sleep(150 * 1000) // Allow ample time for completion
    sc.stop()
  }

  implicit class RichRDD[T](rdd: org.apache.spark.rdd.RDD[T]) {
    def foreachPartitionAsync(f: Iterator[T] => Future[Unit])(implicit ec: ExecutionContext): Unit = {
      rdd.foreachPartition(iterator => f(iterator))
    }
  }
}

          producer.send(message, new CompletionListener {
            override def onCompletion(message: javax.jms.Message): Unit = {
              sentCount.incrementAndGet()
              latch.countDown()
              // Optionally log successful send
              // println(s"Async send completed for: ${message.getBody(classOf[String])}")
            }

            override def onException(message: javax.jms.Message, exception: Exception): Unit = {
              System.err.println(s"Error sending message: ${exception.getMessage} for: ${message.getBody(classOf[String])}")
              latch.countDown()
              // Handle the error appropriately (e.g., retry, logging)
            }
          })
        }

        // Wait for all messages in the batch to be sent (or to fail)
        latch.await()
        println(s"Finished initiating asynchronous send for batch of ${messageBatch.size} messages. ${sentCount.get()} succeeded.")

      } catch {
        case e: Exception => println(s"Error sending batch asynchronously: ${e.getMessage}")
      } finally {
        if (producer != null) producer.close()
        if (session != null) session.close()
        if (connection != null) connection.close()
      }
    }

    println("Initiated asynchronous sending for all batches.")
    sc.stop()
  }
}

Explanation of Asynchronous Sending:

    CompletionListener: The core of asynchronous JMS is the CompletionListener interface. When you call producer.send(message, listener), the send method returns immediately, and the actual delivery of the message happens in the background. The JMS provider then calls either the onCompletion method if the send is successful or the onException method if an error occurs.
    CountDownLatch for Batch Tracking: To know when all messages in a batch have been attempted to be sent (either successfully or with an error), we use a CountDownLatch. It's initialized with the number of messages in the batch. Each time onCompletion or onException is called for a message, the latch is decremented. The latch.await() call blocks until the latch count reaches zero, indicating that all send operations for the current batch have completed (or at least attempted).
    AtomicInteger for Success Count: We use an AtomicInteger to keep track of the number of successfully sent messages within each batch in a thread-safe manner.
    DeliveryMode: The producer.setDeliveryMode() method allows you to specify whether messages should be persistent (DeliveryMode.PERSISTENT) or non-persistent (DeliveryMode.NON_PERSISTENT). Choose based on your reliability requirements. Non-persistent messages are faster but are lost if the broker crashes before they are consumed.
    Error Handling in CompletionListener: The onException method in the CompletionListener is crucial for handling send failures. You should implement appropriate error logging and potentially retry mechanisms here.
    Acknowledgment Mode: The example uses Session.AUTO_ACKNOWLEDGE. For asynchronous sending, especially with potential failures, you might consider Session.CLIENT_ACKNOWLEDGE or JMS transactions (session.beginTransaction(), session.commit(), session.rollback()) for more robust message delivery guarantees. However, these add complexity.

import org.apache.activemq.ActiveMQConnectionFactory
import javax.jms.{Session, TextMessage, MessageProducer}
import org.apache.spark.SparkContext
import org.apache.spark.SparkConf
import scala.concurrent.{Future, ExecutionContext}
import scala.util.{Success, Failure}
import java.util.concurrent.Executors

object AsyncActiveMQProducerBatched {
  def main(args: Array[String]): Unit = {
    val conf = new SparkConf().setAppName("AsyncActiveMQProducerBatched").setMaster("local[*]") // Adjust master
    val sc = new SparkContext(conf)
    implicit val ec = ExecutionContext.fromExecutor(Executors.newFixedThreadPool(5)) // Thread pool for async ops

    val activeMQBrokerURL = "tcp://localhost:61616" // Replace with your ActiveMQ broker URL
    val queueName = "myAsyncQueue"
    val numberOfMessages = 50000
    val batchSize = 100 // Adjust batch size carefully due to message size

    // Generate large message content (50K string)
    val largeMessageContent = (1 to 50 * 1024).map(_ => 'A').mkString

    // 1. Generate RDD of message batches
    val messageBatchesRDD = sc.parallelize(1 to numberOfMessages)
      .grouped(batchSize)
      .map(_.map(i => s"Message $i: $largeMessageContent"))

    println(s"Generating and asynchronously sending $numberOfMessages messages (50K each) in batches of $batchSize to $queueName")

    messageBatchesRDD.foreachPartitionAsync { partitionOfBatches =>
      Future { // Execute processing for each partition asynchronously
        val connectionFactory = new ActiveMQConnectionFactory(activeMQBrokerURL)
        var connection: javax.jms.Connection = null
        var session: Session = null
        var producer: MessageProducer = null

        try {
          connection = connectionFactory.createConnection()
          connection.start()
          session = connection.createSession(false, Session.AUTO_ACKNOWLEDGE)
          val destination = session.createQueue(queueName)
          producer = session.createProducer(destination)

          partitionOfBatches.foreach { messageBatch =>
            val futures = messageBatch.map { messageText =>
              Future { // Asynchronously send each message
                val message = session.createTextMessage(messageText)
                producer.send(message)
              }
            }

            // Optionally, wait for all futures in the batch to complete
            Future.sequence(futures).onComplete {
              case Success(_) => println(s"Batch of ${messageBatch.size} sent asynchronously from partition.")
              case Failure(e) => println(s"Error sending batch asynchronously from partition: ${e.getMessage}")
            }
            // Introduce a small delay to avoid overwhelming ActiveMQ (adjust as needed)
            Thread.sleep(10)
          }
        } catch {
          case e: Exception => println(s"Error in partition processing: ${e.getMessage}")
        } finally {
          if (producer != null) producer.close()
          if (session != null) session.close()
          if (connection != null) connection.close()
        }
      }
    }

    // Keep the main thread alive for a while to allow asynchronous operations to complete
    Thread.sleep(60 * 1000) // Adjust based on the expected time to send all messages

    println("Asynchronous sending initiated. Check logs for completion.")
    sc.stop()
  }

  // Extension method for asynchronous foreachPartition
  implicit class RichRDD[T](rdd: org.apache.spark.rdd.RDD[T]) {
    def foreachPartitionAsync(f: Iterator[T] => Future[Unit])(implicit ec: ExecutionContext): Unit = {
      rdd.foreachPartition(iterator => f(iterator))
    }
  }
}

Here's a sample implementation demonstrating how to connect to the ActiveMQ JMX server and retrieve the QueueSize metric for a specific queue using Spark Scala.

Dependencies:

libraryDependencies += "javax.management" % "javax.management-api" % "1.0.3" // Or a compatible version

Sample Implementation:
Scala

import javax.management.remote.{JMXConnector, JMXConnectorFactory, JMXServiceURL}
import javax.management.{ObjectName}
import org.apache.spark.SparkContext
import org.apache.spark.SparkConf

object ActiveMQJMXMonitor {
  def main(args: Array[String]): Unit = {
    val conf = new SparkConf().setAppName("ActiveMQJMXMonitor").setMaster("local[*]") // Adjust master
    val sc = new SparkContext(conf)

    val jmxURL = "service:jmx:rmi:///jndi/rmi://localhost:1099/jmxrmi" // Replace with your ActiveMQ JMX URL
    val queueName = "myAsyncQueue" // Replace with the name of the queue you want to monitor

    // Function to fetch the QueueSize
    def getQueueSize(jmxUrl: String, queue: String): Option[Int] = {
      var jmxc: JMXConnector = null
      try {
        val serviceURL = new JMXServiceURL(jmxUrl)
        jmxc = JMXConnectorFactory.connect(serviceURL, null)
        val mbsc = jmxc.getMBeanServerConnection

        val objectName = new ObjectName(s"org.apache.activemq:brokerName=localhost,destinationName=$queue,destinationType=Queue,*" ) // Adjust brokerName if needed
        val mbeans = mbsc.queryMBeans(objectName, null)

        if (mbeans.isEmpty) {
          println(s"Queue '$queue' not found in JMX.")
          None
        } else {
          val mbeanInfo = mbeans.iterator().next().getObjectName
          val queueSize = mbsc.getAttribute(mbeanInfo, "QueueSize").asInstanceOf[java.lang.Integer].intValue()
          Some(queueSize)
        }
      } catch {
        case e: Exception =>
          println(s"Error connecting to JMX or retrieving QueueSize: ${e.getMessage}")
          None
      } finally {
        try {
          if (jmxc != null) jmxc.close()
        } catch {
          case e: Exception => println(s"Error closing JMX connection: ${e.getMessage}")
        }
      }
    }

    // Example of how to use the function
    val currentQueueSize = getQueueSize(jmxURL, queueName)

    currentQueueSize match {
      case Some(size) => println(s"Queue '$queueName' size: $size")
      case None => println(s"Could not retrieve queue size for '$queueName'.")
    }

    sc.stop()
  }
}

Explanation:

    JMX Connection Details:
        jmxURL: This variable holds the URL of the ActiveMQ JMX server. The default for a local ActiveMQ instance is often service:jmx:rmi:///jndi/rmi://localhost:1099/jmxrmi. Make sure to adjust this if your ActiveMQ instance is configured differently.
        queueName: Replace "myAsyncQueue" with the actual name of the queue you want to monitor.

    getQueueSize Function:
        Connect to JMX: It establishes a connection to the JMX server using JMXConnectorFactory.connect().
        Get MBean Server Connection: It retrieves the MBeanServerConnection from the connector, which allows you to interact with the managed beans (MBeans).
        Construct Object Name: It creates an ObjectName to identify the MBean for the specific queue. The pattern org.apache.activemq:brokerName=localhost,destinationName=$queue,destinationType=Queue,* is typical for ActiveMQ. You might need to adjust brokerName=localhost if your broker has a different name. The ,* at the end allows matching any MBean with the specified properties.
        Query MBeans: It queries the MBean server for MBeans matching the constructed ObjectName.
        Handle Queue Not Found: If no MBeans are found for the given queue name, it prints a message and returns None.
        Get Queue Size: If MBeans are found, it retrieves the first matching MBean's ObjectName and then gets the value of the QueueSize attribute using mbsc.getAttribute(). The result is cast to java.lang.Integer and then converted to an Int.
        Error Handling: The try-catch blocks handle potential exceptions during JMX connection or attribute retrieval.
        Close Connection: The finally block ensures that the JMX connection is closed to release resources.

    Usage Example:
        The main function calls getQueueSize to retrieve the current size of the specified queue.
        It then uses a match statement to handle the Option[Int] result, printing the queue size or an error message if the size couldn't be retrieved.

Integrating with Your Asynchronous Producer:

You can integrate this getQueueSize function into your asynchronous producer logic. For example, you could periodically check the queue size before sending more batches or individual messages. However, be mindful of the following:

    Polling Overhead: Frequent polling of JMX metrics can introduce overhead on the ActiveMQ server. Choose a reasonable polling interval.
    Consistency: The JMX metrics provide a snapshot of the queue state at the time of the query. There might be a slight delay between the query and the actual state when your producer attempts to send.
    Backpressure Handling: While you can monitor the queue size, it's generally better to rely on ActiveMQ's built-in backpressure mechanisms and handle potential send failures in your asynchronous send callbacks/futures. Using the queue size to prevent sending might be less efficient than letting ActiveMQ manage its resources.
